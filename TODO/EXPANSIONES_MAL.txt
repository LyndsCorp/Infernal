#!/bin/infernal

#----------------------
# Configuracion
global DEBUG = true

#-----------------------

function debuguear(str)
    if DEBUG == true then
        error(str)
    else
        return
    fi
fi

scriptDir = here()

global home = bash $scriptDir/utils/home.sh
global configDir = $home + "/.config/monojo-infernal-teacher-cli"
global clasesDir = $scriptDir + "/clases"
global utilsDir =  $scriptDir + "/utils"

debuguear(clasesDir)
debuguear(configDir)

function bienvenida()
    success("¡Hola, soy Monojo, y hoy te voy a enseñar a programar en Infernal!")

    @decrararname
    name = input("Dime, ¿cómo te llamas?: ")
    local confirmar = input("Te llamas " + name + ", ¿confirmas? (S/n)")
    local confirmacion = lower(confirmar)
    
    if confirmacion == "n" then
        repeat @declararname
    else
        mkdir -p $configDir
        touch $configDir/username.txt
        echo $name > $configDir/username.txt
        return true
    fi
fi

function ejecucion()
    debuguear($utilsDir + "/existeArchivo.sh")
    existeArchivoScript = $utilsDir + "/existeArchivo.sh"
    local existe = bash $existeArchivoScript $configDir/username.txt
    if existe == "false" then
        bienvenida()
    else
        print("Hola")
        #leerDatos()
    fi
fi

if DEBUG == true then
    printAllVars()
fi

ejecucion()
